import React from "react";
import { AbsoluteFill, Easing, interpolate, OffthreadVideo, Sequence, staticFile, useCurrentFrame } from "remotion";

const INK = "#17191f";
const PAPER = "#f5f2eb";
const RED = "#c93a40";
const BRASS = "#e2c79c";
const MINCHO = '"Hiragino Mincho ProN", "Hiragino Mincho Pro", serif';
const GOTHIC = '"Hiragino Sans", "Hiragino Kaku Gothic ProN", sans-serif';

const ease = Easing.bezier(0.2, 0.7, 0.2, 1);
const fadeIn = (f: number, start = 0, dur = 14) => interpolate(f, [start, start + dur], [0, 1], { extrapolateLeft: "clamp", extrapolateRight: "clamp", easing: ease });
const fadeOut = (f: number, end: number, dur = 10) => interpolate(f, [end - dur, end], [1, 0], { extrapolateLeft: "clamp", extrapolateRight: "clamp" });

const Backdrop: React.FC = () => (
  <AbsoluteFill style={{ background: `linear-gradient(160deg, #22252c 0%, ${INK} 55%, #121318 100%)` }}>
    <svg width="1920" height="1080" style={{ position: "absolute", inset: 0, opacity: 0.5 }}>
      <path d="M1180 0H1320L1000 1080H860Z M1460 0H1600L1280 1080H1140Z" fill="#ffffff" opacity="0.03" />
      <path d="M1500 -200L1980 40L1520 1300L1040 1060Z" fill="none" stroke={BRASS} strokeOpacity="0.25" />
    </svg>
  </AbsoluteFill>
);

const Logo: React.FC<{ size?: number }> = ({ size = 40 }) => (
  <div style={{ display: "flex", alignItems: "center", gap: size * 0.3, color: "white", fontFamily: GOTHIC, fontWeight: 700, fontSize: size }}>
    <span style={{ width: size * 0.42, height: size * 0.42, borderRadius: "50%", background: RED, display: "inline-block" }} />
    Tryline
  </div>
);

const Caption: React.FC<{ title: string; sub?: string; frame: number; end: number; x?: number; y?: number; width?: number }> = ({ title, sub, frame, end, x = 120, y = 90, width = 1680 }) => {
  const o = fadeIn(frame, 2) * fadeOut(frame, end);
  const ty = interpolate(fadeIn(frame, 2, 18), [0, 1], [24, 0]);
  return (
    <div style={{ position: "absolute", left: x, top: y, width, opacity: o, transform: `translateY(${ty}px)` }}>
      <div style={{ width: 72, height: 5, background: BRASS, marginBottom: 22 }} />
      <div style={{ fontFamily: MINCHO, fontWeight: 600, fontSize: 64, color: "white", lineHeight: 1.25, whiteSpace: "pre-line" }}>{title}</div>
      {sub ? <div style={{ fontFamily: GOTHIC, fontSize: 28, color: "#d6d0c4", marginTop: 16 }}>{sub}</div> : null}
    </div>
  );
};

const Browser: React.FC<{ src: string; frame: number; end: number; top?: number; width?: number; startFrom?: number }> = ({ src, frame, end, top = 290, width = 1280, startFrom = 0 }) => {
  const h = width * (900 / 1600);
  const o = fadeIn(frame, 0, 12) * fadeOut(frame, end);
  const s = interpolate(fadeIn(frame, 0, 22), [0, 1], [0.96, 1]);
  return (
    <div style={{ position: "absolute", left: (1920 - width) / 2, top, width, opacity: o, transform: `scale(${s})`, transformOrigin: "center top", borderRadius: 14, overflow: "hidden", boxShadow: "0 40px 90px rgba(0,0,0,.55)", border: "1px solid rgba(255,255,255,.12)" }}>
      <div style={{ height: 38, background: "#2a2d34", display: "flex", alignItems: "center", gap: 9, paddingLeft: 18 }}>
        {["#ff5f57", "#febc2e", "#28c840"].map((c) => <span key={c} style={{ width: 13, height: 13, borderRadius: "50%", background: c, display: "inline-block" }} />)}
        <span style={{ marginLeft: 22, fontFamily: GOTHIC, fontSize: 17, color: "#b9b3a8", background: "#1d1f25", padding: "4px 18px", borderRadius: 7 }}>trylinerugby.com</span>
      </div>
      <OffthreadVideo src={staticFile(src)} startFrom={startFrom} muted style={{ width, height: h, display: "block", background: PAPER }} />
    </div>
  );
};

const Phone: React.FC<{ src: string; frame: number; end: number; x: number; top?: number; height?: number; delay?: number }> = ({ src, frame, end, x, top = 300, height = 720, delay = 0 }) => {
  const w = height * (660 / 1434);
  const o = fadeIn(frame, delay, 12) * fadeOut(frame, end);
  const ty = interpolate(fadeIn(frame, delay, 20), [0, 1], [40, 0]);
  return (
    <div style={{ position: "absolute", left: x, top, opacity: o, transform: `translateY(${ty}px)`, padding: 12, background: "#0b0c10", borderRadius: 64, boxShadow: "0 40px 90px rgba(0,0,0,.6)", border: "1px solid rgba(255,255,255,.14)" }}>
      <div style={{ width: w, height, borderRadius: 52, overflow: "hidden" }}>
        <Sequence from={delay} layout="none"><OffthreadVideo src={staticFile(src)} muted style={{ width: w, height, display: "block" }} /></Sequence>
      </div>
    </div>
  );
};

const Title: React.FC = () => {
  const f = useCurrentFrame();
  return (
    <AbsoluteFill style={{ alignItems: "center", justifyContent: "center", opacity: fadeOut(f, 75) }}>
      <div style={{ opacity: fadeIn(f, 0), transform: `translateY(${interpolate(fadeIn(f, 0, 20), [0, 1], [20, 0])}px)` }}><Logo size={46} /></div>
      {["海外ラグビーを、", "日本語で。"].map((line, i) => (
        <div key={line} style={{ fontFamily: MINCHO, fontWeight: 600, fontSize: 104, color: i === 1 ? BRASS : "white", marginTop: i === 0 ? 40 : 0, lineHeight: 1.25, opacity: fadeIn(f, 10 + i * 6, 14), clipPath: `inset(0 0 ${100 - 100 * fadeIn(f, 10 + i * 6, 16)}% 0)`, transform: `translateY(${interpolate(fadeIn(f, 10 + i * 6, 16), [0, 1], [28, 0])}px)` }}>{line}</div>
      ))}
    </AbsoluteFill>
  );
};

const EndCard: React.FC = () => {
  const f = useCurrentFrame();
  return (
    <AbsoluteFill style={{ alignItems: "center", justifyContent: "center" }}>
      <div style={{ opacity: fadeIn(f, 0) }}><Logo size={64} /></div>
      <div style={{ fontFamily: MINCHO, fontWeight: 600, fontSize: 60, color: "white", marginTop: 36, opacity: fadeIn(f, 6) }}>今週の海外ラグビーを、日本時間で追う。</div>
      <div style={{ display: "flex", gap: 28, marginTop: 44, opacity: fadeIn(f, 12) }}>
        <div style={{ fontFamily: GOTHIC, fontWeight: 700, fontSize: 32, color: INK, background: PAPER, padding: "16px 34px", borderRadius: 999 }}>trylinerugby.com</div>
        <div style={{ fontFamily: GOTHIC, fontWeight: 700, fontSize: 32, color: "white", background: RED, padding: "16px 34px", borderRadius: 999 }}>App Store で配信中</div>
      </div>
    </AbsoluteFill>
  );
};

const Scene: React.FC<{ dur: number; children: (f: number, end: number) => React.ReactNode }> = ({ dur, children }) => {
  const f = useCurrentFrame();
  return <AbsoluteFill>{children(f, dur)}</AbsoluteFill>;
};
